/* ============================================================
   draw.js — لوح الرسم الإلكتروني: ألوان، حجم فرشاة، ممحاة، تراجع، مسح، حفظ، إنهاء
   NIG.Draw.open(title) → وعد يتحقق عند الضغط على «تم»
   ============================================================ */
window.NIG = window.NIG || {};
NIG.Draw = (function () {
  const COLORS = ['#d62828', '#ffffff', '#111111', '#1d4ed8', '#15803d', '#d99a06', '#7e22ce', '#f97316'];

  function open(title) {
    return new Promise(res => {
      const w = document.createElement('div'); w.className = 'dr-wrap';
      w.innerHTML = `
        <div class="dr-top"><b>🎨 ${title || 'لوح الرسم'}</b><button class="btn primary dr-done">تم ✔</button></div>
        <div class="dr-tools">
          <div class="dr-colors">${COLORS.map((c, i) => `<button class="dr-c${i === 0 ? ' on' : ''}" data-c="${c}" style="background:${c}" aria-label="لون"></button>`).join('')}</div>
          <label class="dr-size">الحجم <input type="range" min="2" max="40" value="8"></label>
          <button class="dr-b" data-a="eraser">🧽 ممحاة</button>
          <button class="dr-b" data-a="undo">↩ تراجع</button>
          <button class="dr-b" data-a="clear">🗑 مسح</button>
          <button class="dr-b" data-a="save">💾 حفظ</button>
        </div>
        <canvas class="dr-canvas"></canvas>`;
      document.body.appendChild(w);
      const cv = w.querySelector('canvas'), g = cv.getContext('2d');
      const size = w.querySelector('input'); let color = COLORS[0], erase = false, drawing = false, hist = [], last;
      function fit() {
        const r = cv.getBoundingClientRect(), d = window.devicePixelRatio || 1;
        const keep = cv.width ? g.getImageData(0, 0, cv.width, cv.height) : null;
        cv.width = r.width * d; cv.height = r.height * d;
        g.fillStyle = '#fff'; g.fillRect(0, 0, cv.width, cv.height);
        if (keep) g.putImageData(keep, 0, 0);
        g.lineCap = g.lineJoin = 'round';
      }
      fit(); hist = [g.getImageData(0, 0, cv.width, cv.height)];
      const pos = e => { const r = cv.getBoundingClientRect(); return [(e.clientX - r.left) * cv.width / r.width, (e.clientY - r.top) * cv.height / r.height]; };
      cv.onpointerdown = e => { drawing = true; cv.setPointerCapture(e.pointerId); last = pos(e); stroke(last, last); e.preventDefault(); };
      cv.onpointermove = e => { if (!drawing) return; const p = pos(e); stroke(last, p); last = p; };
      const end = () => { if (!drawing) return; drawing = false; hist.push(g.getImageData(0, 0, cv.width, cv.height)); if (hist.length > 30) hist.shift(); };
      cv.onpointerup = cv.onpointercancel = end;
      function stroke(a, b) {
        g.strokeStyle = erase ? '#fff' : color; g.lineWidth = size.value * (window.devicePixelRatio || 1);
        g.beginPath(); g.moveTo(a[0], a[1]); g.lineTo(b[0] + .01, b[1]); g.stroke();
      }
      w.querySelectorAll('.dr-c').forEach(b => b.onclick = () => {
        color = b.dataset.c; erase = false;
        w.querySelectorAll('.dr-c').forEach(x => x.classList.remove('on')); b.classList.add('on');
        w.querySelector('[data-a=eraser]').classList.remove('on');
      });
      w.querySelectorAll('.dr-b').forEach(b => b.onclick = () => {
        const a = b.dataset.a;
        if (a === 'eraser') { erase = !erase; b.classList.toggle('on', erase); }
        if (a === 'undo' && hist.length > 1) { hist.pop(); g.putImageData(hist[hist.length - 1], 0, 0); }
        if (a === 'clear') { g.fillStyle = '#fff'; g.fillRect(0, 0, cv.width, cv.height); hist.push(g.getImageData(0, 0, cv.width, cv.height)); }
        if (a === 'save') {
          try {
            const l = document.createElement('a'); l.download = 'رسمتي.png'; l.href = cv.toDataURL('image/png'); l.click();
          } catch (e) {}
        }
      });
      w.querySelector('.dr-done').onclick = () => { w.remove(); res(true); };
    });
  }
  return { open };
})();

/* ============================================================
   fx.js — المؤثرات: أصوات (قابلة للإغلاق)، الولد الكرتوني، الصوت، ورود الإجابة الصحيحة
   كل الأصوات مُولَّدة بالكود (لا تحتاج ملفات ولا إنترنت)
   ============================================================ */
window.NIG = window.NIG || {};
NIG.FX = (function () {
  let ctx = null, on = true;
  try { on = localStorage.getItem('nig_sound') !== '0'; } catch (e) {}

  function ac() {
    if (!ctx) { try { ctx = new (window.AudioContext || window.webkitAudioContext)(); } catch (e) { return null; } }
    if (ctx.state === 'suspended') ctx.resume();
    return ctx;
  }
  /* المتصفحات (خصوصًا الآيفون) تمنع الصوت حتى أول لمسة: نفتحه عند أول ضغطة */
  function unlock() {
    const c = ac(); if (!c) return;
    try { const b = c.createBuffer(1, 1, 22050), s = c.createBufferSource(); s.buffer = b; s.connect(c.destination); s.start(0); } catch (e) {}
    if ('speechSynthesis' in window) { try { speechSynthesis.getVoices(); } catch (e) {} }
  }
  ['pointerdown', 'touchstart', 'click', 'keydown'].forEach(ev => document.addEventListener(ev, unlock, { passive: true }));

  function tone(f, t0, d, type, vol) {
    const c = ac(); if (!c) return;
    const o = c.createOscillator(), g = c.createGain();
    o.type = type || 'sine'; o.frequency.value = f;
    const t = c.currentTime + t0;
    g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(Math.min(1, (vol || 0.2) * 2), t + 0.02);
    g.gain.exponentialRampToValueAtTime(0.0001, t + d);
    o.connect(g); g.connect(c.destination); o.start(t); o.stop(t + d + 0.05);
  }
  /* ---------- مؤثرات صوتية: نُولّد ملف WAV بالكود ونشغّله بعنصر audio (أوثق على الهواتف) ---------- */
  const SR = 22050;
  // نوتة: [تردد, بداية(ث), مدة(ث), موجة, شدة]   ضجيج تصفيق: ['n', بداية, مدة, شدة]
  const N = (f, t, d, w, v) => [f, t, d, w || 'tri', v || .5];
  const SPEC = {
    right: [ // فانفار حماسي + تصفيق
      N(523, 0, .14), N(659, .12, .14), N(784, .24, .14), N(1046, .36, .45, 'sq', .45),
      N(784, .36, .45, 'tri', .35), N(1318, .62, .5, 'tri', .5), N(1568, .8, .55, 'tri', .4),
      ['n', .5, .9, .5]],
    wrong: [ // «وووومب وووومب» حزين
      N(330, 0, .3, 'saw', .5), N(311, .3, .3, 'saw', .5), N(262, .62, .6, 'saw', .55), N(131, .62, .6, 'sq', .35)],
    dice: [N(300, 0, .06, 'sq', .3), N(390, .08, .06, 'sq', .3), N(480, .16, .06, 'sq', .3), N(600, .24, .08, 'sq', .3)],
    step: [N(520, 0, .07, 'tri', .4)],
    flip: [N(400, 0, .08, 'sine', .5), N(700, .07, .12, 'sine', .45)],
    ladder: [N(400, 0, .1, 'sine', .5), N(500, .08, .1, 'sine', .5), N(650, .16, .1, 'sine', .5), N(850, .24, .25, 'sine', .5)],
    win: [N(523, 0, .2), N(659, .18, .2), N(784, .36, .2), N(1046, .54, .3), N(784, .8, .15), N(1046, .95, .15), N(1318, 1.1, .7, 'sq', .45), ['n', .9, 1.3, .5]]
  };
  const wave = (w, ph) => w === 'sq' ? (Math.sin(ph) >= 0 ? 1 : -1) * .6 : w === 'saw' ? ((ph / (2 * Math.PI)) % 1) * 2 - 1 : w === 'tri' ? 2 / Math.PI * Math.asin(Math.sin(ph)) : Math.sin(ph);
  function render(name) {
    const spec = SPEC[name]; let len = 0;
    spec.forEach(x => { const t = x[0] === 'n' ? x[1] + x[2] : x[1] + x[2]; len = Math.max(len, t); });
    const n = Math.ceil((len + .1) * SR), buf = new Float32Array(n);
    spec.forEach(x => {
      if (x[0] === 'n') { // تصفيق: ضجيج متقطع
        const i0 = Math.floor(x[1] * SR), i1 = Math.min(n, Math.floor((x[1] + x[2]) * SR));
        for (let i = i0; i < i1; i++) {
          const t = (i - i0) / SR, env = Math.min(1, t * 20) * Math.max(0, 1 - t / x[2]);
          const clap = (Math.floor(t * 14) % 2 ? 1 : .55);
          buf[i] += (Math.random() * 2 - 1) * x[3] * .5 * env * clap;
        }
        return;
      }
      const [f, t0, d, w, v] = x, i0 = Math.floor(t0 * SR), cnt = Math.floor(d * SR);
      for (let k = 0; k < cnt && i0 + k < n; k++) {
        const t = k / cnt, env = Math.min(1, k / (SR * .01)) * Math.pow(1 - t, 1.3);
        buf[i0 + k] += wave(w, 2 * Math.PI * f * k / SR) * v * env;
      }
    });
    let mx = 0; for (let i = 0; i < n; i++) mx = Math.max(mx, Math.abs(buf[i]));
    const g = mx > .95 ? .95 / mx : 1;
    const bytes = new Uint8Array(44 + n * 2), dv = new DataView(bytes.buffer);
    const wr = (o, str) => { for (let i = 0; i < str.length; i++) dv.setUint8(o + i, str.charCodeAt(i)); };
    wr(0, 'RIFF'); dv.setUint32(4, 36 + n * 2, true); wr(8, 'WAVEfmt '); dv.setUint32(16, 16, true); dv.setUint16(20, 1, true);
    dv.setUint16(22, 1, true); dv.setUint32(24, SR, true); dv.setUint32(28, SR * 2, true); dv.setUint16(32, 2, true); dv.setUint16(34, 16, true);
    wr(36, 'data'); dv.setUint32(40, n * 2, true);
    for (let i = 0; i < n; i++) dv.setInt16(44 + i * 2, Math.max(-1, Math.min(1, buf[i] * g)) * 32767, true);
    let bin = ''; for (let i = 0; i < bytes.length; i += 8192) bin += String.fromCharCode.apply(null, bytes.subarray(i, i + 8192));
    return 'data:audio/wav;base64,' + btoa(bin);
  }
  const cache = {};
  function sfx(name) {
    if (!on || !SPEC[name]) return;
    try {
      if (!cache[name]) cache[name] = render(name);
      const a = new Audio(cache[name]); a.volume = 1;
      const pr = a.play();
      if (pr && pr.catch) pr.catch(() => { try { webSfx(name); } catch (e) {} });
    } catch (e) { try { webSfx(name); } catch (e2) {} }
  }
  /* احتياطي: WebAudio */
  function webSfx(name) {
    SPEC[name].forEach(x => { if (x[0] !== 'n') tone(x[0], x[1], x[2], x[3] === 'sq' ? 'square' : x[3] === 'saw' ? 'sawtooth' : x[3] === 'tri' ? 'triangle' : 'sine', x[4] * .4); });
  }

  /* ---------- الصوت البشري (صوت الجهاز العربي، نبرة طفولية) ---------- */
  function speak(text) {
    if (!on || !('speechSynthesis' in window)) return;
    try {
      speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      const vs = speechSynthesis.getVoices();
      const v = vs.find(x => /ar[-_]EG/i.test(x.lang)) || vs.find(x => /^ar/i.test(x.lang));
      if (v) u.voice = v;
      u.lang = v ? v.lang : 'ar-EG'; u.pitch = 1.7; u.rate = 1.05; u.volume = 1;
      speechSynthesis.speak(u);
    } catch (e) {}
  }

  /* ---------- الولد الكرتوني (SVG) ---------- */
  const BOY = (mood) => `
  <svg viewBox="0 0 120 140" width="100%" height="100%" aria-hidden="true">
    <ellipse cx="60" cy="132" rx="30" ry="6" fill="rgba(0,0,0,.2)"/>
    <path d="M30 128c0-26 14-40 30-40s30 14 30 40z" fill="#d62828"/>
    <rect x="52" y="78" width="16" height="14" rx="6" fill="#f2b987"/>
    <circle cx="60" cy="52" r="32" fill="#f7c9a0"/>
    <path d="M27 48c-2-24 16-34 33-34s35 10 33 34c-6-12-18-16-33-16S33 36 27 48z" fill="#2b1a10"/>
    <circle cx="28" cy="56" r="6" fill="#f7c9a0"/><circle cx="92" cy="56" r="6" fill="#f7c9a0"/>
    <circle cx="48" cy="54" r="5" fill="#fff"/><circle cx="72" cy="54" r="5" fill="#fff"/>
    <circle cx="${mood ? 49 : 48}" cy="55" r="2.6" fill="#222"/><circle cx="${mood ? 73 : 72}" cy="55" r="2.6" fill="#222"/>
    ${mood
      ? '<path d="M44 68q16 18 32 0z" fill="#fff" stroke="#8a2b2b" stroke-width="2.5" stroke-linejoin="round"/><path d="M42 44q6-5 12 0M66 44q6-5 12 0" stroke="#2b1a10" stroke-width="3" fill="none" stroke-linecap="round"/>'
      : '<path d="M46 76q14-10 28 0" stroke="#8a2b2b" stroke-width="3" fill="none" stroke-linecap="round"/><path d="M42 46q6 4 12 0M66 46q6 4 12 0" stroke="#2b1a10" stroke-width="3" fill="none" stroke-linecap="round"/>'}
    <circle cx="38" cy="66" r="5" fill="#f3a38f" opacity=".6"/><circle cx="82" cy="66" r="5" fill="#f3a38f" opacity=".6"/>
    ${mood ? '<path d="M92 92l16-22" stroke="#f7c9a0" stroke-width="9" stroke-linecap="round"/><circle cx="109" cy="68" r="7" fill="#f7c9a0"/>' : ''}
  </svg>`;

  let boyTimer = null;
  const GOOD = ['أحسنت يا بطل!'];
  const BAD = ['حظ أوفر المرة القادمة!'];
  const pick = a => a[Math.floor(Math.random() * a.length)];

  function boy(ok, text) {
    let el = document.getElementById('fx-boy');
    if (!el) { el = document.createElement('div'); el.id = 'fx-boy'; document.body.appendChild(el); }
    text = text || pick(ok ? GOOD : BAD);
    el.className = 'fx-boy show ' + (ok ? 'happy' : 'sad');
    el.innerHTML = `<div class="fx-bubble">${text}</div><div class="fx-fig">${BOY(ok)}</div>`;
    setTimeout(() => speak(text), 700);
    clearTimeout(boyTimer); boyTimer = setTimeout(() => el.classList.remove('show'), 2600);
  }

  /* ---------- ورود حمراء تتساقط ---------- */
  function roses() {
    const box = document.createElement('div'); box.className = 'fx-roses';
    let h = '';
    for (let i = 0; i < 18; i++) {
      h += `<i style="left:${Math.random() * 96}%;animation-delay:${Math.random() * .8}s;animation-duration:${2 + Math.random() * 1.5}s;font-size:${22 + Math.random() * 18}px">🌹</i>`;
    }
    box.innerHTML = h; document.body.appendChild(box);
    setTimeout(() => box.remove(), 4200);
  }

  /* تغذية راجعة كاملة: صوت + ولد + (ورود إن صح) */
  function feedback(ok, text) {
    sfx(ok ? 'right' : 'wrong'); boy(ok, text); if (ok) roses();
    const fl = document.createElement('div'); fl.className = 'fx-flash ' + (ok ? 'g' : 'r'); document.body.appendChild(fl); setTimeout(() => fl.remove(), 700);
  }

  /* ---------- زر الصوت ---------- */
  function init() {
    if (document.getElementById('fx-toggle')) return;
    const b = document.createElement('button');
    b.id = 'fx-toggle'; b.className = 'fx-toggle'; b.type = 'button';
    const paint = () => { b.textContent = on ? '🔊 الصوت' : '🔇 صامت'; b.title = on ? 'إيقاف الصوت' : 'تشغيل الصوت'; };
    b.onclick = () => {
      on = !on; try { localStorage.setItem('nig_sound', on ? '1' : '0'); } catch (e) {}
      if (!on && 'speechSynthesis' in window) speechSynthesis.cancel();
      paint(); if (on) { sfx('right'); speak('الصوت شغّال يا بطل'); }
    };
    paint();
    document.body.appendChild(b);
    if ('speechSynthesis' in window) speechSynthesis.getVoices();
  }
  document.addEventListener('DOMContentLoaded', init);
  if (document.readyState !== 'loading') init();

  return { state: () => (ctx ? ctx.state : 'none'), sfx, speak, boy, roses, feedback, init, isOn: () => on };
})();

   cardui.js — سحب الكروت المقلوبة، عرض الكارت الأصلي، الأسئلة والتحديات
   play(ev, player) تُرجع وعدًا بنتيجة: { points, move, swap, correct }
   ============================================================ */
window.NIG = window.NIG || {};

NIG.CardUI = (function () {
  const $ = id => document.getElementById(id);
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const esc = s => String(s).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));
  const shuffle = a => { a = a.slice(); for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } return a; };

  const ALL = ['info', 'flag', 'values', 'history', 'geography', 'figures', 'pharaohs', 'culture', 'battles'];
  const POOL = {
    situation: ['values'], history: ['history'], geography: ['geography'],
    figures: ['figures', 'pharaohs'], culture: ['culture'], battles: ['battles'],
    info: ['info'], flag: ['flag'], challenge: ['challenge'],
    quiz: ALL, question: ALL
  };
  const MIXED = { quiz: 1, question: 1 };
  const PENALTY = { 'belong-2': 5, 'message-8': 3, 'hero-7': 2 };   // كما هو مكتوب على الكروت
  const LETTERS = ['أ', 'ب', 'ج', 'د'];

  /* ---------- اختيار الكروت (بدون تكرار داخل المباراة) ---------- */
  function candidates(type) {
    const S = NIG.Game.get();
    S.used = S.used || {};
    const key = MIXED[type] ? 'mixed' : type;
    const all = NIG.CARDS.filter(c => POOL[type].includes(c.cat));
    let used = S.used[key] || [];
    let free = all.filter(c => !used.includes(c.id));
    if (!free.length) { used = []; S.used[key] = used; free = all; }
    return { free, key };
  }
  function markUsed(key, id) {
    const S = NIG.Game.get();
    (S.used[key] = S.used[key] || []).push(id);
    NIG.Game.save();
  }

  function backOf(card) {
    const b = card.cat === 'pharaohs' ? 'figures' : card.cat;
    return NIG.BACKS[b];
  }

  /* ---------- أدوات الواجهة ---------- */
  function layer(html) { $('card-body').innerHTML = html; $('card-layer').classList.remove('hidden'); $('card-layer').scrollTop = 0; }
  function closeLayer() { $('card-layer').classList.add('hidden'); $('card-body').innerHTML = ''; }
  function waitClick(sel) {
    return new Promise(res => { const b = document.querySelector(sel); b.onclick = () => res(); });
  }
  const head = (t, sub) => `<div class="cl-head"><b>${esc(t)}</b>${sub ? `<span>${esc(sub)}</span>` : ''}</div>`;

  /* ---------- 1) اختيار كارت مقلوب ---------- */
  async function pickCard(type, who) {
    const { free, key } = candidates(type);
    const n = Math.min(NIG.RULES.cardChoices || 6, free.length);
    const slots = shuffle(free).slice(0, n);
    const mixed = !!MIXED[type];
    const label = NIG.TYPES[type].label;
    const tiles = slots.map((c, i) => `
      <button class="fc" data-i="${i}" aria-label="كارت ${i + 1}">
        <div class="fci">
          <div class="fcb">${mixed ? '<span class="qmark">؟</span>' : `<img src="${backOf(c)}" alt="ظهر الكارت" draggable="false">`}</div>
          <div class="fcf"><img src="${c.img}" alt="" draggable="false"></div>
        </div></button>`).join('');
    layer(`${head('محطة ' + label, who)}
      <p class="cl-hint">اختاري كارتًا واحدًا لتكتشفي ما بداخله 👇</p>
      <div class="fc-grid n${n}">${tiles}</div>`);
    const idx = await new Promise(res => {
      document.querySelectorAll('.fc').forEach(b => b.onclick = () => res(+b.dataset.i));
    });
    document.querySelectorAll('.fc').forEach(b => { b.disabled = true; if (+b.dataset.i !== idx) b.classList.add('fade'); });
    if (NIG.FX) NIG.FX.sfx('flip');
    document.querySelector(`.fc[data-i="${idx}"]`).classList.add('on');
    const card = slots[idx];
    markUsed(key, card.id);
    await sleep(1100);
    return card;
  }

  /* ---------- 2) عرض الكارت الأصلي ---------- */
  async function revealCard(card, btnText, label) {
    layer(`${head(label, '')}
      <div class="reveal"><img class="big-card pop" src="${card.img}" alt="الكارت"></div>
      <div class="cl-actions"><button id="cl-go" class="btn primary">${btnText}</button></div>`);
    await waitClick('#cl-go');
  }

  /* ---------- 3) سؤال اختيار من متعدد ---------- */
  function askMCQ(card, qtext, options, pts, opts) {
    return new Promise(res => {
      const right = options[0];
      const list = shuffle(options);
      layer(`
        <div class="qv">
          <button class="qthumb-btn" id="q-zoom" title="تكبير الكارت"><img class="qthumb" src="${card.img}" alt="الكارت"></button>
          <div class="qbox">
            <div class="q-label">${esc(opts.label || 'السؤال')}</div>
            <p class="qtext">${esc(qtext)}</p>
            <div class="opts">${list.map((t, i) => `<button class="opt" data-t="${esc(t)}"><i>${LETTERS[i]}</i><span>${esc(t)}</span></button>`).join('')}</div>
            <div id="q-fb" class="fb hidden"></div>
            <div class="cl-actions"><button id="q-next" class="btn primary hidden">متابعة ◀</button></div>
          </div>
        </div>`);
      $('q-zoom').onclick = () => {
        const z = document.createElement('div'); z.className = 'zoom-view';
        z.innerHTML = `<img src="${card.img}" alt="">`; z.onclick = () => z.remove(); $('card-body').appendChild(z);
      };
      let done = false;
      document.querySelectorAll('.opt').forEach(b => b.onclick = () => {
        if (done) return; done = true;
        const ok = b.dataset.t === right;
        document.querySelectorAll('.opt').forEach(x => {
          x.disabled = true;
          if (x.dataset.t === right) x.classList.add('right');
        });
        if (!ok) b.classList.add('wrong');
        if (NIG.FX) NIG.FX.feedback(ok);
        const fb = $('q-fb'); fb.classList.remove('hidden'); fb.classList.add(ok ? 'good' : 'bad');
        fb.innerHTML = ok
          ? `<b>✔ إجابة صحيحة! +${pts} نقطة</b>${opts.okNote ? `<small>${esc(opts.okNote)}</small>` : ''}`
          : `<b>✘ إجابة غير صحيحة</b><small>الإجابة الصحيحة: ${esc(right)}${opts.badNote ? ' — ' + esc(opts.badNote) : ''}</small>`;
        const nx = $('q-next'); nx.classList.remove('hidden'); nx.onclick = () => res(ok);
        nx.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      });
    });
  }

  /* ---------- 4) تحدٍّ مفتوح (نشاط) ---------- */
  function openChallenge(card, penalty, swapText) {
    return new Promise(res => {
      layer(`
        <div class="qv">
          <button class="qthumb-btn" id="q-zoom"><img class="qthumb" src="${card.img}" alt="الكارت"></button>
          <div class="qbox">
            <div class="q-label">${esc(card.title)}</div>
            <p class="qtext">${esc(card.task)}</p>
            ${card.draw ? '<button id="draw-open" class="btn ghost draw-open">🎨 افتح لوح الرسم</button>' : ''}
            ${card.sec ? `<div class="timer"><span id="tm">${card.sec}</span><small>ثانية</small><button id="tm-go" class="btn ghost">ابدأ المؤقت ⏱</button></div>` : ''}
            <p class="soft">بعد تنفيذ التحدي اختاروا النتيجة (يقرّرها الحَكَم: المعلمة أو الفريق المنافس):</p>
            <div class="cl-actions two">
              <button id="ch-ok" class="btn primary">✔ أنجزنا التحدي (+${card.pts})</button>
              <button id="ch-no" class="btn danger">✘ لم ننجز (${swapText || (penalty === 1 ? 'ارجع خطوة' : 'ارجع ' + penalty + ' خطوات')})</button>
            </div>
          </div>
        </div>`);
      $('q-zoom').onclick = () => {
        const z = document.createElement('div'); z.className = 'zoom-view';
        z.innerHTML = `<img src="${card.img}" alt="">`; z.onclick = () => z.remove(); $('card-body').appendChild(z);
      };
      const dro = $('draw-open'); if (dro) dro.onclick = () => NIG.Draw.open(card.title);
      let timer = null;
      const stop = () => { if (timer) clearInterval(timer); };
      const tg = $('tm-go');
      if (tg) tg.onclick = () => {
        stop(); let t = card.sec; $('tm').textContent = t; tg.disabled = true;
        timer = setInterval(() => {
          t--; const el = $('tm'); if (!el) return stop();
          el.textContent = t; if (t <= 5) el.classList.add('low');
          if (t <= 0) { stop(); el.textContent = '⏰'; tg.disabled = false; tg.textContent = 'إعادة المؤقت'; }
        }, 1000);
      };
      $('ch-ok').onclick = () => { stop(); res(true); };
      $('ch-no').onclick = () => { stop(); res(false); };
    });
  }

  /* ---------- نتيجة قصيرة بعد التحدي ---------- */
  async function toast(html) {
    layer(`<div class="result-box">${html}<div class="cl-actions"><button id="cl-go" class="btn primary">متابعة ◀</button></div></div>`);
    if (NIG.FX && /🌟|💪/.test(html)) NIG.FX.feedback(/🌟/.test(html));
    await waitClick('#cl-go');
  }

  /* ---------- المسار الكامل ---------- */
  async function play(ev, player) {
    try { return await playInner(ev, player); }
    finally { closeLayer(); }          // دائمًا نُغلق طبقة الكروت ليظهر اللوح
  }

  async function playInner(ev, player) {
    const S = NIG.Game.get();
    const type = ev.type;
    const who = (S.mode === 'team' && player.members > 1) ? `${player.name} — العضو ${player.member + 1}` : player.name;
    const label = 'محطة ' + NIG.TYPES[type].label;
    const card = await pickCard(type, who);
    let result = { points: 0, move: 0, swap: false, correct: null };

    if (card.cat === 'challenge') {
      const penalty = PENALTY[card.num] || 1;
      let ok;
      if (card.kind === 'mcq') {
        await revealCard(card, 'ابدأ التحدي ◀', label);
        ok = await askMCQ(card, card.task, card.o, card.pts, { label: card.title });
      } else {
        await revealCard(card, 'ابدأ التحدي ◀', label);
        const team = S.mode === 'team' && player.members > 1;
        ok = await openChallenge(card, penalty, card.swap ? (team ? 'بدّلوا العضو' : 'ارجع خطوة') : null);
      }
      result.correct = ok;
      if (ok) result.points = card.pts;
      else if (card.swap && S.mode === 'team' && player.members > 1) result.swap = true;
      else result.move = -(card.swap ? 1 : penalty);
      if (card.kind !== 'mcq') {
        await toast(ok
          ? `<div class="big">🌟</div><h3>أحسنتم! +${card.pts} نقطة</h3>`
          : `<div class="big">💪</div><h3>حاولوا في المرة القادمة</h3><p class="soft">${result.swap ? 'يُبدَّل العضو الذي يلعب.' : 'تعودون ' + (-result.move === 1 ? 'خطوة' : (-result.move) + ' خطوات') + ' للخلف.'}</p>`);
      }
      return result;
    }

    // كروت المعلومات: اقرأ الكارت ثم أجب
    await revealCard(card, 'اقرأت الكارت — إلى السؤال ◀', label);
    const pts = NIG.RULES.pointsCorrect || 10;
    const ok = await askMCQ(card, card.q, card.o, pts, { label: 'السؤال' });
    result.correct = ok;
    if (ok) { result.points = pts; result.move = NIG.RULES.stepOnCorrect || 0; }
    else { result.move = -(NIG.RULES.stepOnWrong || 0); }
    return result;
  }

  function end() { closeLayer(); }

  return { play, end, _closeLayer: closeLayer };
})();

/* ============================================================
   ui.js — الشاشات والنرد وحركة القطع والاحتفال
   ============================================================ */
(function () {
  const G = NIG.Game;
  const $ = id => document.getElementById(id);
  const sleep = ms => new Promise(r => setTimeout(r, ms));

  const START_POS = { x: 735, y: 837 };      // مكان القطع قبل البداية
  let setupMode = 'solo';
  let setupCount = 2;
  let busy = false;
  let zoomIdx = 0;
  const ZOOMS = [100, 150, 200, 260];

  /* ---------------- الشاشات ---------------- */
  function show(id) {
    document.querySelectorAll('.screen').forEach(s => s.classList.toggle('active', s.id === id));
    window.scrollTo(0, 0);
  }

  /* ---------------- نافذة عامة ---------------- */
  function modal(html, actions) {
    return new Promise(resolve => {
      $('modal-body').innerHTML = html;
      const box = $('modal-actions'); box.innerHTML = '';
      actions.forEach(a => {
        const b = document.createElement('button');
        b.className = 'btn ' + (a.cls || 'primary'); b.textContent = a.text;
        b.onclick = () => { $('modal').classList.add('hidden'); resolve(a.value); };
        box.appendChild(b);
      });
      $('modal').classList.remove('hidden');
    });
  }

  /* ---------------- إعداد اللاعبين ---------------- */
  function renderSetup() {
    const team = setupMode === 'team';
    $('setup-title').textContent = team ? 'إعداد الفرق' : 'إعداد الطلاب';
    const box = $('setup-fields'); box.innerHTML = '';
    for (let i = 0; i < setupCount; i++) {
      const c = NIG.COLORS[i];
      const row = document.createElement('div'); row.className = 'field';
      row.innerHTML =
        `<label>${team ? 'اسم الفريق' : 'اسم الطالب'} ${i + 1}</label>` +
        `<input type="text" maxlength="24" id="f-name-${i}" placeholder="${team ? 'مثال: فريق النيل' : 'اكتب الاسم'}">` +
        (team ? `<label>عدد الأعضاء</label><input type="number" min="1" max="12" value="4" id="f-mem-${i}">` : '') +
        `<label>اللون</label><div class="swatches" id="f-col-${i}" data-val="${c.id}"></div>`;
      box.appendChild(row);
      const sw = row.querySelector('.swatches');
      NIG.COLORS.forEach(col => {
        const b = document.createElement('button');
        b.type = 'button'; b.className = 'sw' + (col.id === c.id ? ' on' : '');
        b.style.background = col.hex; b.title = col.name;
        b.onclick = () => { sw.dataset.val = col.id; sw.querySelectorAll('.sw').forEach(x => x.classList.remove('on')); b.classList.add('on'); };
        sw.appendChild(b);
      });
    }
  }

  function startFromSetup() {
    const entries = [];
    const used = new Set();
    for (let i = 0; i < setupCount; i++) {
      const colId = $('f-col-' + i).dataset.val;
      if (used.has(colId)) { alert('اختاري لونًا مختلفًا لكل ' + (setupMode === 'team' ? 'فريق' : 'طالب')); return; }
      used.add(colId);
      const col = NIG.COLORS.find(c => c.id === colId);
      let name = $('f-name-' + i).value.trim() || ((setupMode === 'team' ? 'الفريق ' : 'الطالب ') + (i + 1));
      entries.push({
        name, color: col.hex,
        members: setupMode === 'team' ? Math.max(1, Math.min(12, parseInt($('f-mem-' + i).value) || 1)) : 1
      });
    }
    G.create(setupMode, entries);
    openGame();
  }

  /* ---------------- اللوحة والقطع ---------------- */
  const pct = (x, y) => ({ l: (x / NIG.BOARD.w * 100), t: (y / NIG.BOARD.h * 100) });

  function pawnSVG(color) {
    return `<svg viewBox="0 0 40 56" xmlns="http://www.w3.org/2000/svg">
      <ellipse cx="20" cy="52" rx="15" ry="4" fill="rgba(0,0,0,.35)"/>
      <path d="M20 14 C10 24 6 40 4 48 L36 48 C34 40 30 24 20 14Z" fill="${color}" stroke="#fff" stroke-width="2.5"/>
      <circle cx="20" cy="11" r="9" fill="${color}" stroke="#fff" stroke-width="2.5"/>
      <circle cx="17" cy="8" r="2.6" fill="rgba(255,255,255,.55)"/></svg>`;
  }

  function slotOffset(idx, total) {
    if (total <= 1) return { dx: 0, dy: 0 };
    const a = (idx / total) * Math.PI * 2 - Math.PI / 2;
    return { dx: Math.cos(a) * 12, dy: Math.sin(a) * 8 };
  }

  function stationXY(pos) {
    if (pos <= 0) return START_POS;
    const s = G.byNum(pos); return s || START_POS;
  }

  function placeToken(p, instant) {
    const el = document.querySelector(`.token[data-id="${p.id}"]`);
    if (!el) return;
    const st = stationXY(p.pos);
    const sameTile = G.get().players.filter(q => q.pos === p.pos);
    const o = slotOffset(sameTile.indexOf(p), sameTile.length);
    const c = pct(st.x + o.dx, st.y + o.dy);
    if (instant) el.style.transition = 'none';
    el.style.left = c.l + '%'; el.style.top = c.t + '%';
    if (instant) { void el.offsetWidth; el.style.transition = ''; }
  }

  function refreshTokens(instant) {
    G.get().players.forEach(p => placeToken(p, instant));
  }

  function buildTokens() {
    const box = $('tokens'); box.innerHTML = '';
    G.get().players.forEach(p => {
      const d = document.createElement('div');
      d.className = 'token'; d.dataset.id = p.id;
      d.innerHTML = pawnSVG(p.color);
      d.title = p.name;
      box.appendChild(d);
    });
    refreshTokens(true);
  }

  function renderScoreboard() {
    const S = G.get();
    $('scoreboard').innerHTML = S.players.map((p, i) =>
      `<div class="chip ${i === S.turn ? 'now' : ''}" style="--c:${p.color}">
         <i class="dot"></i><b>${esc(p.name)}</b>
         <span>المحطة ${p.pos <= 0 ? 'البداية' : (p.pos >= G.FINISH ? 'النهاية' : p.pos)}</span>
         <em>⭐ ${p.score}</em></div>`).join('');
  }

  function renderTurn() {
    const S = G.get(), p = G.current();
    const who = (S.mode === 'team' && p.members > 1) ? ` — العضو ${p.member + 1}` : '';
    $('turn-banner').innerHTML = `<i class="dot" style="background:${p.color}"></i> دور: <b>${esc(p.name)}</b>${who}`;
    renderScoreboard();
  }

  function esc(s) { return String(s).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m])); }
  function say(t) { $('msg').textContent = t; }

  function applyZoom() {
    $('board').style.width = ZOOMS[zoomIdx] + '%';
    focusCurrent(false);
  }
  function focusCurrent(smooth) {
    const p = G.current(); if (!p) return;
    const sc = $('board-scroll'), b = $('board');
    const st = stationXY(p.pos);
    const x = st.x / NIG.BOARD.w * b.clientWidth;
    const y = st.y / NIG.BOARD.h * b.clientHeight;
    sc.scrollTo({ left: x - sc.clientWidth / 2, top: y - sc.clientHeight / 2, behavior: smooth ? 'smooth' : 'auto' });
  }

  /* ---------------- النرد ---------------- */
  const FACE_ROT = { 1: [0, 0], 6: [0, 180], 2: [0, -90], 5: [0, 90], 3: [-90, 0], 4: [90, 0] };
  let spinBase = 0;

  function buildDice() {
    const cube = $('cube'); cube.innerHTML = '';
    const pips = { 1: [4], 2: [0, 8], 3: [0, 4, 8], 4: [0, 2, 6, 8], 5: [0, 2, 4, 6, 8], 6: [0, 2, 3, 5, 6, 8] };
    const place = { 1: 'rotateY(0deg)', 6: 'rotateY(180deg)', 2: 'rotateY(90deg)', 5: 'rotateY(-90deg)', 3: 'rotateX(90deg)', 4: 'rotateX(-90deg)' };
    for (let f = 1; f <= 6; f++) {
      const face = document.createElement('div');
      face.className = 'face'; face.style.transform = place[f] + ' translateZ(var(--half))';
      for (let i = 0; i < 9; i++) {
        const c = document.createElement('i');
        if (pips[f].includes(i)) c.className = 'pip';
        face.appendChild(c);
      }
      cube.appendChild(face);
    }
    showFace(1, true);
  }
  function showFace(n, instant) {
    const cube = $('cube'); const [rx, ry] = FACE_ROT[n];
    spinBase += 720;
    cube.style.transition = instant ? 'none' : 'transform 1s cubic-bezier(.2,.8,.2,1)';
    cube.style.transform = `rotateX(${rx + (instant ? 0 : spinBase)}deg) rotateY(${ry + (instant ? 0 : spinBase)}deg)`;
  }

  async function onRoll() {
    if (NIG.FX) NIG.FX.sfx('dice');
    if (busy || G.get().over) return;
    busy = true; $('dice').classList.add('off');
    const p = G.current();
    const n = G.roll();
    say(`${p.name} يرمي النرد…`);
    showFace(n, false);
    await sleep(1050);
    say(`خرج الرقم ${n}`);
    await sleep(350);

    await walk(p, Math.min(G.FINISH, p.pos + n));
    await resolveLanding(p);
  }

  /* تحريك القطعة محطة بمحطة */
  async function walk(p, target) {
    const steps = G.path(p.pos, target);
    for (const n of steps) {
      G.moveTo(p, n);
      placeToken(p);
      const el = document.querySelector(`.token[data-id="${p.id}"]`);
      el.classList.add('hop'); setTimeout(() => el.classList.remove('hop'), 260);
      renderScoreboard(); focusCurrent(true);
      say(`المحطة ${n}`); if (NIG.FX) NIG.FX.sfx('step');
      await sleep(300);
    }
    refreshTokens();
  }

  async function slideTo(p, target) {
    G.moveTo(p, target);
    placeToken(p); renderScoreboard(); focusCurrent(true);
    await sleep(900);
    refreshTokens();
  }

  async function resolveLanding(p) {
    const ev = G.landing(p);
    switch (ev.kind) {
      case 'finish': return win(p);
      case 'ladder':
        if (NIG.FX) NIG.FX.sfx('ladder');
        say(ev.up ? '🪜 سلم! اطلع إلى المحطة ' + ev.to : '🪜 أنزل السلم إلى المحطة ' + ev.to);
        await sleep(700);
        await slideTo(p, ev.to);
        break;
      case 'forward':
        say(`👏 أحسنت! تقدّم ${ev.steps === 1 ? 'خطوة' : 'خطوتين'}`);
        await sleep(600);
        await walk(p, Math.min(G.FINISH, p.pos + ev.steps));
        break;
      case 'back':
        say(`↩ ارجع ${ev.steps === 1 ? 'خطوة' : 'خطوتين'}`);
        await sleep(600);
        await walk(p, Math.max(1, p.pos - ev.steps));
        break;
      case 'swap':
        await modal(`<div class="big">🔄</div><h3>محطة تبديل</h3><p>${G.get().mode === 'team' ? 'بدّلوا العضو الذي يلعب في الفريق.' : 'لا يوجد أعضاء في اللعب الفردي — تابعي.'}</p>`,
          [{ text: 'تابع', value: 1 }]);
        if (G.get().mode === 'team') G.rotateMember(p);
        break;
      case 'card': {
        let r = {};
        if (NIG.CardUI) r = await NIG.CardUI.play(ev, p); else await cardPlaceholder(ev);
        await applyCardResult(p, r);
        break;
      }
    }
    if (G.hasWon(p)) return win(p);
    endTurn();
  }

  /* تطبيق نتيجة الكارت: نقاط + حركة + تبديل عضو */
  async function applyCardResult(p, r) {
    if (r.points) { p.score += r.points; G.save(); renderScoreboard(); }
    if (r.swap) { G.rotateMember(p); renderTurn(); }
    if (r.move) {
      const target = Math.max(1, Math.min(G.FINISH, p.pos + r.move));
      if (target !== p.pos) {
        say(r.move > 0 ? `👏 تقدّم ${r.move === 1 ? 'خطوة' : r.move + ' خطوات'}` : `↩ ارجع ${r.move === -1 ? 'خطوة' : (-r.move) + ' خطوات'}`);
        await sleep(500);
        await walk(p, target);
      }
    }
  }

  /* احتياطي إن لم يُحمَّل ملف الكروت */
  async function cardPlaceholder(ev) {
    const t = NIG.TYPES[ev.type].label;

      [{ text: 'تابع', value: 1 }]);
  }

  function endTurn() {

    show('screen-win'); startConfetti();

  }

/* ============================================================
   game.js — منطق اللعبة (بدون أي واجهة): الأدوار، النرد، الحركة، السلالم
   ============================================================ */
window.NIG = window.NIG || {};

NIG.Game = (function () {
  const FINISH = NIG.STATIONS[NIG.STATIONS.length - 1].n; // 50
  const SAVE_KEY = 'nig_save_v1';

  let state = null;

  const byNum = n => NIG.STATIONS.find(s => s.n === n);

  function create(mode, entries) {
    // entries: [{name, color, members}]
    state = {
      mode,
      turn: 0,
      round: 1,
      over: false,
      used: {},            // الكروت التي ظهرت في هذه المباراة (لمنع التكرار)
      players: entries.map((e, i) => ({
        id: i,
        name: e.name,
        color: e.color,
        members: e.members || 1,
        member: 0,           // رقم العضو الحالي داخل الفريق
        pos: 0,              // 0 = قبل البداية
        score: 0
      })),
      log: []
    };
    save();
    return state;
  }

  const get = () => state;
  const current = () => state.players[state.turn];

  function roll() { return 1 + Math.floor(Math.random() * 6); }

  /* يحسب مسار الخطوات (محطة بمحطة) من موضع إلى آخر */
  function path(from, to) {
    const p = [];
    const dir = to >= from ? 1 : -1;
    for (let n = from + dir; dir > 0 ? n <= to : n >= to; n += dir) p.push(n);
    return p;
  }

  /* يطبق أثر المحطة بعد الوصول. يُرجع وصفًا للحدث */
  function landing(player) {
    const st = byNum(player.pos);
    if (!st) return { kind: 'none' };
    if (player.pos >= FINISH) return { kind: 'finish' };

    if (NIG.RULES.useLadders) {
      const l = NIG.LADDERS.find(l => l.from === st.n);
      if (l) return { kind: 'ladder', from: l.from, to: l.to, up: l.to > l.from, station: st };
    }
    if (st.t === 'forward') return { kind: 'forward', steps: st.steps, station: st };
    if (st.t === 'back')    return { kind: 'back', steps: st.steps, station: st };
    if (st.t === 'swap')    return { kind: 'swap', station: st };
    if (NIG.TYPES[st.t] && NIG.TYPES[st.t].card) return { kind: 'card', type: st.t, station: st };
    return { kind: 'none', station: st };
  }

  /* تحريك اللاعب إلى موضع جديد (دون تطبيق أثر المحطة) */
  function moveTo(player, n) {
    player.pos = Math.max(0, Math.min(FINISH, n));
    save();
  }

  function nextTurn() {
    const p = current();
    if (state.mode === 'team' && p.members > 1) p.member = (p.member + 1) % p.members;
    state.turn = (state.turn + 1) % state.players.length;
    if (state.turn === 0) state.round++;
    save();
    return current();
  }

  function rotateMember(player) {
    if (player.members > 1) player.member = (player.member + 1) % player.members;
    save();
  }

  /* الفائز: من وصل النهاية */
  function hasWon(player) {
    return NIG.RULES.exactFinish ? player.pos === FINISH : player.pos >= FINISH;
  }

  /* ---------- الحفظ المحلي (يعمل بدون إنترنت) ---------- */
  function save() {
    try { localStorage.setItem(SAVE_KEY, JSON.stringify(state)); } catch (e) { /* تجاهل */ }
  }
  function load() {
    try {
      const raw = localStorage.getItem(SAVE_KEY);
      if (!raw) return null;
      const s = JSON.parse(raw);
      if (!s || s.over || !Array.isArray(s.players)) return null;
      state = s; return state;
    } catch (e) { return null; }
  }
  function hasSave() {
    try {
      const raw = localStorage.getItem(SAVE_KEY);
      if (!raw) return false;
      const s = JSON.parse(raw);
      return !!(s && !s.over && s.players);
    } catch (e) { return false; }
  }
  function clearSave() { try { localStorage.removeItem(SAVE_KEY); } catch (e) {} }
  function finish() { state.over = true; save(); }

  return { FINISH, byNum, create, get, current, roll, path, landing, moveTo,
           nextTurn, rotateMember, hasWon, load, hasSave, clearSave, finish, save };
})();
